# Зависимости для Raspberry Pi 5 (aarch64)
#
# PyQt6 ставится ИЗ PIP, а не из apt:
#   • на PyPI есть wheel manylinux для ARM64 (aarch64);
#   • только pip-версия PyQt6 содержит модули QtQml / QtQuick /
#     QtQuickControls2, которые нужны QML-интерфейсу;
#   • системный apt-пакет python3-pyqt6 их НЕ содержит, поэтому запуск
#     падает с ModuleNotFoundError: No module named 'PyQt6.QtQml'.
#
# Запустите install.sh — он сам создаст venv и установит всё нужное.

# Связка PyQt6 + Qt-runtime одной линейки 6.7.x, manylinux_2_28 (aarch64):
# ставится на любом Raspberry Pi OS (glibc >= 2.28) БЕЗ сборки из исходников
# и содержит все QML-модули (QtQuick, QtQuick.Controls, стиль Material).
PyQt6==6.7.1
PyQt6-Qt6==6.7.3

PyYAML>=6.0.0
QtAwesome>=1.3.0
rpi-lgpio>=0.4.0
